ki_parser_content = """import os
import sexpdata

def load_sexpr(filepath: str):
    if not os.path.exists(filepath):
        raise FileNotFoundError(f"File not found: {filepath}")
    with open(filepath, "r", encoding="utf-8") as f:
        content = f.read()
    return sexpdata.loads(content)

def parse_schematic(filepath: str) -> dict:
    try:
        tree = load_sexpr(filepath)
    except Exception as e:
        return {"error": str(e), "file": filepath}
    
    components = {}
    nets = set()
    title_block = {}
    
    def traverse(node):
        if not isinstance(node, list) or not node:
            return
        
        tag = node[0]
        if isinstance(tag, sexpdata.Symbol):
            tag_name = tag.value()
            
            if tag_name == "title":
                if len(node) > 1:
                    title_block["title"] = str(node[1])
            elif tag_name == "company":
                if len(node) > 1:
                    title_block["company"] = str(node[1])
            elif tag_name == "rev":
                if len(node) > 1:
                    title_block["revision"] = str(node[1])
            elif tag_name == "symbol":
                # Find reference designator and value inside symbol properties
                ref = "UNKNOWN"
                val = "UNKNOWN"
                for item in node[1:]:
                    if isinstance(item, list) and len(item) > 0:
                        if isinstance(item[0], sexpdata.Symbol) and item[0].value() == "property":
                            if len(item) > 2 and str(item[1]) == "Reference":
                                ref = str(item[2])
                            elif len(item) > 2 and str(item[1]) == "Value":
                                val = str(item[2])
                if ref != "UNKNOWN":
                    components[ref] = {"value": val, "type": "symbol"}
            elif tag_name == "net":
                if len(node) > 2:
                    nets.add(str(node[2]))
                    
        for item in node:
            if isinstance(item, list):
                traverse(item)

    traverse(tree)
    
    return {
        "filename": os.path.basename(filepath),
        "title_block": title_block,
        "component_count": len(components),
        "components": components,
        "net_count": len(nets),
        "nets": sorted(list(nets))
    }

def parse_pcb(filepath: str) -> dict:
    try:
        tree = load_sexpr(filepath)
    except Exception as e:
        return {"error": str(e), "file": filepath}
        
    modules = []
    nets = {}
    
    def traverse(node):
        if not isinstance(node, list) or not node:
            return
        tag = node[0]
        if isinstance(tag, sexpdata.Symbol):
            tag_name = tag.value()
            if tag_name == "footprint":
                fp_name = str(node[1]) if len(node) > 1 else "Unknown"
                ref = "UNKNOWN"
                for item in node[1:]:
                    if isinstance(item, list) and len(item) > 0:
                        if isinstance(item[0], sexpdata.Symbol) and item[0].value() == "property":
                            if len(item) > 2 and str(item[1]) == "Reference":
                                ref = str(item[2])
                modules.append({"reference": ref, "footprint": fp_name})
            elif tag_name == "net":
                if len(node) > 2:
                    net_id = str(node[1])
                    net_name = str(node[2])
                    nets[net_id] = net_name
                    
        for item in node:
            if isinstance(item, list):
                traverse(item)
                
    traverse(tree)
    
    return {
        "filename": os.path.basename(filepath),
        "footprint_count": len(modules),
        "footprints": modules,
        "net_count": len(nets),
        "nets": nets
    }
"""

with open(os.path.join(parsers_dir, "ki_parser.py"), "w") as f:
    f.write(ki_parser_content)

# 3. kicad_cli.py
kicad_cli_content = """import sys
import json
import argparse
from parsers.ki_parser import parse_schematic, parse_pcb

def main():
    parser = argparse.ArgumentParser(description="KiCad Analyzer CLI for Hardware Agent")
    subparsers = parser.add_subparsers(dest="command", required=True)
    
    # Inventory subcommand
    inv_parser = subparsers.add_parser("inventory", help="Extract components and summary from schematic or PCB")
    inv_parser.add_argument("file", help="Path to .kicad_sch or .kicad_pcb file")
    
    # Netlist subcommand
    net_parser = subparsers.add_parser("netlist", help="Extract netlist and connections")
    net_parser.add_argument("file", help="Path to KiCad file")
    
    # DRC Lite subcommand
    drc_parser = subparsers.add_parser("drc-lite", help="Perform basic schematic-to-net checks")
    drc_parser.add_argument("file", help="Path to KiCad schematic or PCB")
    
    args = parser.parse_args()
    
    if args.file.endswith(".kicad_sch"):
        data = parse_schematic(args.file)
    elif args.file.endswith(".kicad_pcb"):
        data = parse_pcb(args.file)
    else:
        print(json.dumps({"error": "Unsupported file extension. Use .kicad_sch or .kicad_pcb"}))
        sys.exit(1)
        
    if args.command == "inventory":
        print(json.dumps(data, indent=2))
    elif args.command == "netlist":
        print(json.dumps({"nets": data.get("nets", data.get("nets", {}))}, indent=2))
    elif args.command == "drc-lite":
        # Basic check: verify components found and nets exist
        issues = []
        if data.get("component_count", 0) == 0 and data.get("footprint_count", 0) == 0:
            issues.append("Warning: No components or footprints detected.")
        print(json.dumps({"status": "passed" if not issues else "warnings", "issues": issues, "summary": data}, indent=2))

if __name__ == "__main__":
    main()
"""

with open(os.path.join(kicad_dir, "kicad_cli.py"), "w") as f:
    f.write(kicad_cli_content)

    if not os.path.exists(filepath):

            elif tag_name == "symbol":
                ref, val = "UNKNOWN", "UNKNOWN"
